  module testbench;

reg clk;
reg reset;
reg tx_start;
reg [7:0] data_in;

wire tick;
wire tx;
wire [7:0] data_out;
wire data_ready;

always #5 clk = ~clk;

baud_gen bg(
    .clk(clk),
    .reset(reset),
    .tick(tick)
);

uart_tx tx1(
    .clk(clk),
    .reset(reset),
    .tick(tick),
    .tx_start(tx_start),
    .data_in(data_in),
    .tx(tx)
);

uart_rx rx1(
    .clk(clk),
    .reset(reset),
    .tick(tick),
    .rx(tx),
    .data_out(data_out),
    .data_ready(data_ready)
);

initial
begin

    clk = 0;
    reset = 1;
    tx_start = 0;
    data_in = 0;

    $dumpfile("dump.vcd");
    $dumpvars;

    #20;
    reset = 0;

    #20;

    data_in = 8'b10101010;

    tx_start = 1;

    #50;
    tx_start = 0;

    #5000;

    $finish;

end

endmodule
